import React, { useState } from 'react';
import {
  View,
  Text,
  TextInput,
  TouchableOpacity,
  StyleSheet,
} from 'react-native';
import { Audio } from 'expo-av';

const CORRECT_KEY = 'dev123';

export default function App() {
  const [key, setKey] = useState('');
  const [isDeveloper, setIsDeveloper] = useState(false);
  const [sound, setSound] = useState(null);

  async function playMoan() {
    if (sound) {
      await sound.stopAsync();
      await sound.unloadAsync();
    }
    try {
      const { sound: newSound } = await Audio.Sound.createAsync(
        require('./assets/moan.mp3')
      );
      setSound(newSound);
      await newSound.playAsync();
    } catch (error) {
      console.log('Ошибка воспроизведения звука', error);
    }
  }

  async function handleLogin(useKey) {
    if (useKey && key.trim() === CORRECT_KEY) {
      if (sound) {
        await sound.stopAsync();
      }
      setIsDeveloper(true);
    } else {
      playMoan();
    }
  }

  if (isDeveloper) {
    return (
      <View style={styles.container}>
        <Text style={styles.devTitle}>👨‍💻 Режим разработчика</Text>
        <Text style={styles.devSub}>Ты вошёл как создатель. Стоны отключены.</Text>
        <TouchableOpacity
          style={styles.logoutBtn}
          onPress={() => setIsDeveloper(false)}>
          <Text>Выйти</Text>
        </TouchableOpacity>
      </View>
    );
  }

  return (
    <View style={styles.container}>
      <Text style={styles.title}>HelloGuys</Text>
      <TextInput
        style={styles.input}
        placeholder="Введите ключ разработчика"
        value={key}
        onChangeText={setKey}
        secureTextEntry
      />
      <TouchableOpacity
        style={styles.keyBtn}
        onPress={() => handleLogin(true)}>
        <Text style={styles.btnText}>Войти с ключом</Text>
      </TouchableOpacity>

      <TouchableOpacity
        style={styles.noKeyBtn}
        onPress={() => handleLogin(false)}>
        <Text style={styles.btnText}>Войти БЕЗ ключа</Text>
      </TouchableOpacity>
    </View>
  );
}

const styles = StyleSheet.create({
  container: {
    flex: 1,
    justifyContent: 'center',
    padding: 30,
    backgroundColor: '#ffffff',
  },
  title: {
    fontSize: 30,
    fontWeight: 'bold',
    textAlign: 'center',
    marginBottom: 30,
    color: '#000',
  },
  input: {
    borderWidth: 1,
    borderColor: '#aaa',
    borderRadius: 10,
    padding: 15,
    fontSize: 18,
    marginBottom: 20,
  },
  keyBtn: {
    backgroundColor: '#4CAF50',
    padding: 15,
    borderRadius: 10,
    marginBottom: 10,
  },
  noKeyBtn: {
    backgroundColor: '#f44336',
    padding: 15,
    borderRadius: 10,
    marginBottom: 20,
  },
  btnText: {
    color: '#fff',
    textAlign: 'center',
    fontSize: 18,
    fontWeight: 'bold',
  },
  devTitle: {
    fontSize: 28,
    textAlign: 'center',
    marginBottom: 10,
    color: '#2196F3',
  },
  devSub: {
    fontSize: 16,
    textAlign: 'center',
    marginBottom: 40,
    color: '#333',
  },
  logoutBtn: {
    alignSelf: 'center',
    backgroundColor: '#eee',
    paddingVertical: 10,
    paddingHorizontal: 30,
    borderRadius: 8,
  },
});
